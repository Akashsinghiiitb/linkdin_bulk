def get_companies_from_internal_api(count = 10):
    try:
        res = requests.get(f"{base_url}/api/company-without-location-headquarter?count={count}&token={token}&v=2", headers={"auth":token,"api":'1'})
        raw = res.json()
        run_id = raw['run_id']
        return raw['data'], run_id
    except Exception as e:
        do_log("Error at get:", e)
        pass
    
def post_companies_to_internal_api(companies):
    try:
        res = requests.post(f"{base_url}/api/save-location-by-company-detail-id", json=companies)
        return res.json()
    except:
        pass
def post_company_details(data, run_id):
    try:
        res = requests.post(f"{base_url}/api/save-CompanyDetails", json=data, headers={"auth":token,"api":'1', "run_id":str(run_id)})
        return res.json()
    except Exception as e:
        do_log("Error while on save API: ",e)
    
def get_values_from_dict(dict):
    return [dict[i] for i in dict]

def get_company_location(company_url, data=None):
    if data == None:
        data = get_company_details(company_url)
    try:
        location_list = get_values_from_dict(data['headquarter'])
        location = " ".join(location_list)
        return location
    except:
        if data == None:
            return None
        return None
